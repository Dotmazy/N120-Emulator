cmake_minimum_required(VERSION 3.24)
project(N120Emulator VERSION 0.4.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_AUTOMOC ON)
include(CTest)
option(N120_BUILD_DESKTOP "Build the existing Qt Widgets desktop application" ON)
option(N120_BUILD_MOBILE "Build the Qt Quick mobile application" OFF)
if(ANDROID)
    set(N120_BUILD_DESKTOP OFF)
    set(N120_BUILD_MOBILE ON)
endif()
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Network Gui)
add_library(n120_core STATIC shared/core/Firmware.cpp)
target_include_directories(n120_core PUBLIC shared)
set_target_properties(n120_core PROPERTIES AUTOMOC OFF)
add_library(n120_interfaces STATIC shared/interfaces/EmulatorBackend.h)
target_link_libraries(n120_interfaces PUBLIC Qt6::Core)
target_include_directories(n120_interfaces PUBLIC shared)
add_library(n120_input STATIC shared/input/KeyboardTranslator.cpp shared/input/KeyboardTranslator.h shared/input/CalculatorLayout.cpp shared/resources/CalculatorStyle.cpp)
target_link_libraries(n120_input PUBLIC Qt6::Gui)
target_include_directories(n120_input PUBLIC shared shared/input)
function(n120_add_calculator_assets target resource_name)
    qt_add_resources(${target} ${resource_name} PREFIX "/" BASE "${PROJECT_SOURCE_DIR}" FILES
        "${PROJECT_SOURCE_DIR}/assets/epsilon-background.png"
        "${PROJECT_SOURCE_DIR}/assets/upsilon-background.png"
        "${PROJECT_SOURCE_DIR}/assets/omega-background.png")
endfunction()
if(NOT ANDROID)
    add_library(n120_renode STATIC backends/renode/RenodeManager.cpp backends/renode/RenodeManager.h)
    target_link_libraries(n120_renode PUBLIC n120_core n120_interfaces Qt6::Network)
    target_include_directories(n120_renode PUBLIC backends/renode)
    add_custom_target(n120_resources ALL COMMAND ${CMAKE_COMMAND} -E copy_directory "${PROJECT_SOURCE_DIR}/renode" "${CMAKE_BINARY_DIR}/renode")
endif()
if(N120_BUILD_DESKTOP)
    add_subdirectory(platforms/windows)
endif()

add_subdirectory(backends/remote)
if(BUILD_TESTING AND NOT ANDROID)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    qt_add_executable(n120_shared_tests tests/shared_tests.cpp)
    target_link_libraries(n120_shared_tests PRIVATE n120_core n120_remote Qt6::Test)
    add_test(NAME shared_contracts COMMAND n120_shared_tests)
endif()
if(N120_BUILD_MOBILE)
    add_subdirectory(platforms/android)
endif()
