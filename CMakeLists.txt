cmake_minimum_required(VERSION 3.24)
project(N120Emulator VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets Network)
qt_add_executable(n120 WIN32 src/main.cpp src/RenodeManager.cpp src/RenodeManager.h src/MainWindow.cpp src/MainWindowActions.cpp src/MainWindow.h src/StartupDialog.cpp src/SettingsDialog.cpp src/CalculatorWidget.cpp src/CalculatorWidget.h src/KeyboardTranslator.cpp src/KeyboardTranslator.h src/LogViewer.cpp src/LogViewer.h)
qt_add_resources(n120 calculator_assets PREFIX "/" FILES assets/calculator.png)
set_target_properties(n120 PROPERTIES OUTPUT_NAME "n120-studio")
target_link_libraries(n120 PRIVATE Qt6::Widgets Qt6::Network)
target_compile_definitions(n120 PRIVATE N120_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
if(MSVC)
  target_compile_options(n120 PRIVATE /W4)
else()
  target_compile_options(n120 PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_custom_target(n120_resources ALL COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/renode" "${CMAKE_CURRENT_BINARY_DIR}/renode")
add_dependencies(n120 n120_resources)
include(CTest)
if(BUILD_TESTING)
 find_package(Qt6 REQUIRED COMPONENTS Test)
 qt_add_executable(n120_desktop_tests tests/desktop_tests.cpp src/CalculatorWidget.cpp src/CalculatorWidget.h src/KeyboardTranslator.cpp src/KeyboardTranslator.h)
 target_include_directories(n120_desktop_tests PRIVATE src)
 target_link_libraries(n120_desktop_tests PRIVATE Qt6::Widgets Qt6::Test)
 qt_add_resources(n120_desktop_tests test_assets PREFIX "/" FILES assets/calculator.png)
 add_test(NAME desktop_controls COMMAND n120_desktop_tests)
 add_test(NAME qt_startup COMMAND n120 --ui-smoke)
 set_tests_properties(qt_startup PROPERTIES TIMEOUT 10)
endif()
