cmake_minimum_required(VERSION 3.24)
project(N120Emulator VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets Network)
qt_add_executable(n120 WIN32 src/main.cpp src/RenodeManager.cpp src/RenodeManager.h src/MainWindow.cpp src/MainWindowActions.cpp src/MainWindow.h src/StartupDialog.cpp src/SettingsDialog.cpp src/CalculatorWidget.cpp src/CalculatorWidget.h src/KeyboardTranslator.cpp src/KeyboardTranslator.h src/LogViewer.cpp src/LogViewer.h src/WindowsUsb.h)
qt_add_resources(n120 calculator_assets PREFIX "/" FILES assets/calculator.png)
set_target_properties(n120 PROPERTIES OUTPUT_NAME "n120-studio")
target_link_libraries(n120 PRIVATE Qt6::Widgets Qt6::Network)
if(MSVC)
  target_compile_options(n120 PRIVATE /W4)
else()
  target_compile_options(n120 PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_custom_target(n120_resources ALL COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/renode" "${CMAKE_CURRENT_BINARY_DIR}/renode")
add_dependencies(n120 n120_resources)
include(CTest)
if(BUILD_TESTING)
 find_package(Qt6 REQUIRED COMPONENTS Test)
 qt_add_executable(n120_desktop_tests tests/desktop_tests.cpp src/CalculatorWidget.cpp src/CalculatorWidget.h src/KeyboardTranslator.cpp src/KeyboardTranslator.h)
 target_include_directories(n120_desktop_tests PRIVATE src)
 target_link_libraries(n120_desktop_tests PRIVATE Qt6::Widgets Qt6::Test)
 qt_add_resources(n120_desktop_tests test_assets PREFIX "/" FILES assets/calculator.png)
 qt_add_executable(n120_manager_tests tests/manager_tests.cpp src/RenodeManager.cpp src/RenodeManager.h)
 target_include_directories(n120_manager_tests PRIVATE src)
 target_link_libraries(n120_manager_tests PRIVATE Qt6::Network Qt6::Test)
 add_test(NAME manager_validation COMMAND n120_manager_tests)
 add_test(NAME desktop_controls COMMAND n120_desktop_tests)
 add_test(NAME qt_startup COMMAND n120 --ui-smoke)
 set_tests_properties(manager_validation desktop_controls qt_startup PROPERTIES TIMEOUT 15)
 set_tests_properties(desktop_controls qt_startup PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()

# Stage only public runtime inputs; Renode, drivers and firmware remain external.
install(TARGETS n120 RUNTIME DESTINATION .)
install(DIRECTORY renode/ DESTINATION renode FILES_MATCHING PATTERN "*.cs" PATTERN "*.repl")
install(FILES README.md THIRD_PARTY_NOTICES.md DESTINATION .)
set(N120_PUBLIC_DOCS
    docs/architecture.md
    docs/building.md
    docs/usb.md
    docs/debugging.md
)
install(FILES ${N120_PUBLIC_DOCS} DESTINATION docs)
foreach(document IN LISTS N120_PUBLIC_DOCS)
    configure_file("${document}" "${CMAKE_CURRENT_BINARY_DIR}/${document}" COPYONLY)
endforeach()
if(WIN32)
    get_target_property(qmake_path Qt6::qmake IMPORTED_LOCATION)
    get_filename_component(qt_bin_dir "${qmake_path}" DIRECTORY)
    find_program(N120_WINDEPLOYQT windeployqt HINTS "${qt_bin_dir}" REQUIRED)
    configure_file(cmake/DeployWindows.cmake.in DeployWindows.cmake @ONLY)
    install(SCRIPT "${CMAKE_CURRENT_BINARY_DIR}/DeployWindows.cmake")
endif()
add_custom_command(TARGET n120 POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/README.md" "$<TARGET_FILE_DIR:n120>/README.md"
    )
