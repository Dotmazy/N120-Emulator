cmake_minimum_required(VERSION 3.24)
project(N120LocalCpuProbe LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(UNICORN_ARCH arm CACHE STRING "" FORCE)
set(UNICORN_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(UNICORN_INSTALL OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(UNICORN_LEGACY_STATIC_ARCHIVE OFF CACHE BOOL "" FORCE)
set(N120_UNICORN_SOURCE "" CACHE PATH "Unicorn 2.1.4 checkout")
if(NOT EXISTS "${N120_UNICORN_SOURCE}/CMakeLists.txt")
    message(FATAL_ERROR "Set N120_UNICORN_SOURCE to Unicorn 2.1.4 source")
endif()
add_subdirectory("${N120_UNICORN_SOURCE}" unicorn)
add_executable(n120-cpu-probe main.cpp ../../../shared/core/Firmware.cpp)
target_include_directories(n120-cpu-probe PRIVATE ../../../shared)
target_link_libraries(n120-cpu-probe PRIVATE unicorn)
